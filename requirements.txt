# AI Sign Language Translator (PSL) dependencies
# Python 3.9 - 3.12 recommended (MediaPipe wheel availability)

# Data handling
numpy
pandas

# Webcam capture + landmark extraction (same format as the dataset)
opencv-python
mediapipe

tensorflow

scikit-learn

# the fallback on other systems.
pyttsx3
edge-tts        # one-time generation of the spoken Urdu clips (src/make_urdu_audio.py, needs internet)

# Plots and notebooks
matplotlib
jupyter
